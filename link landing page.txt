// ============================================================
//  GOOGLE APPS SCRIPT - API đọc sản phẩm từ Google Sheet (bản tối ưu)
//  Cách cập nhật: mở Apps Script -> dán đè toàn bộ code này -> Lưu ->
//  Triển khai > Quản lý bản triển khai > Sửa (biểu tượng bút) > Phiên bản: Mới > Triển khai.
//  (URL /exec giữ nguyên nên không phải đổi gì trong web.)
//  Sau đó chạy hàm setupTriggers() MỘT LẦN và cấp quyền.
//
//  THEO DÕI LƯỢT BẤM (mới):
//    - Trang web gửi "lượt vào" và "lượt bấm sản phẩm" tới doPost() bên dưới.
//    - Dữ liệu ghi vào tab "LuotBam" trong chính Google Sheet này (không có IP, không có tên, không cookie).
//    - Trang kiểm tra trên máy bạn đọc thống kê qua doGet?action=stats&key=... (phải đúng khóa bí mật).
//    - Chạy hàm setupClickTracking() MỘT LẦN để tạo tab, tạo khóa bí mật và cài dọn dẹp tự động.
//    - Sau khi sửa code: Triển khai > Quản lý bản triển khai > Sửa > Phiên bản: Mới > Triển khai.
// ============================================================

// ---- Cấu hình theo dõi lượt bấm ----
var CLICK_SHEET = 'LuotBam';
var CLICK_KEEP_DAYS = 120;          // Giữ nhật ký thô 120 ngày, cũ hơn tự xóa (trang kiểm tra đã lưu tổng hợp trên máy)
var MAX_EVENTS_PER_SESSION = 300;   // Chặn một phiên gửi quá nhiều sự kiện (chống spam)
var MAX_BODY_LENGTH = 500;
var TIMEZONE = 'Asia/Ho_Chi_Minh';
var VALID_TYPES = { view: 1, click: 1 };
var VALID_SOURCES = { tiktok: 1, instagram: 1, facebook: 1, zalo: 1, other: 1 };
var VALID_DEVICES = { ios: 1, android: 1, desktop: 1, other: 1 };

var CACHE_KEY = 'products_v2';
var CACHE_TTL = 21600;       // giây (6 giờ - mức tối đa của CacheService)
var CHUNK_SIZE = 30000;      // ký tự mỗi mảnh. Giới hạn 100KB tính theo BYTE;
                             // tiếng Việt tối đa 3 byte/ký tự nên 30000 ký tự luôn an toàn
var SHEET_NAME = '';         // Điền tên tab chứa sản phẩm (ví dụ 'Sheet1'). Để trống = tab đầu tiên.

// Chỉ giữ những cột web thật sự dùng (bỏ cột lỗi #REF!, cột thừa...)
var WANTED_COLUMNS = ['ID', 'Tên Sản Phẩm', 'Ảnh Sản Phẩm', 'Link Shopee', 'Link Shope'];

function doGet(e) {
  // Yêu cầu thống kê lượt bấm (cần khóa bí mật), còn lại là lấy danh sách sản phẩm như cũ
  if (e && e.parameter && e.parameter.action === 'stats') {
    return statsResponse_(e);
  }

  try {
    var json = getCache_(CACHE_KEY);

    if (!json) {
      json = JSON.stringify(readSheet_());
      try {
        putCache_(CACHE_KEY, json);
      } catch (cacheErr) {
        // Ghi cache lỗi thì vẫn trả dữ liệu bình thường
        console.error('Không ghi được cache: ' + cacheErr.message);
      }
    }
    return ContentService.createTextOutput(json)
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (SHEET_NAME) {
    var named = ss.getSheetByName(SHEET_NAME);
    if (!named) throw new Error('Không tìm thấy tab "' + SHEET_NAME + '"');
    return named;
  }
  return ss.getSheets()[0]; // tab đầu tiên (ổn định hơn getActiveSheet trong web app)
}

function readSheet_() {
  var sheet = getSheet_();
  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();
  if (lastRow < 2 || lastCol < 1) return [];

  var values = sheet.getRange(1, 1, lastRow, lastCol).getValues();
  var headers = values[0];

  // Chỉ lấy vị trí các cột cần dùng
  var wanted = [];
  for (var j = 0; j < headers.length; j++) {
    if (WANTED_COLUMNS.indexOf(String(headers[j]).trim()) !== -1) {
      wanted.push({ index: j, name: String(headers[j]).trim() });
    }
  }

  var idIndex = -1;
  for (var w = 0; w < wanted.length; w++) {
    if (wanted[w].name === 'ID') idIndex = wanted[w].index;
  }

  var result = [];
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    if (idIndex !== -1 && !row[idIndex]) continue; // bỏ dòng trống không có ID
    var obj = {};
    for (var k = 0; k < wanted.length; k++) {
      var cell = row[wanted[k].index];
      // Bỏ giá trị lỗi công thức (#REF!, #N/A...) để không gửi rác xuống web
      if (typeof cell === 'string' && cell.charAt(0) === '#' && /^#[A-Z\/0!?]+$/.test(cell)) cell = '';
      obj[wanted[k].name] = cell;
    }
    result.push(obj);
  }
  return result;
}

// ---- Cache có chia nhỏ (vượt 100KB/key) ----
function putCache_(key, str) {
  var cache = CacheService.getScriptCache();
  var parts = {}, n = Math.ceil(str.length / CHUNK_SIZE);
  for (var i = 0; i < n; i++) parts[key + '_' + i] = str.substr(i * CHUNK_SIZE, CHUNK_SIZE);
  parts[key + '_n'] = String(n);
  cache.putAll(parts, CACHE_TTL);
}

function getCache_(key) {
  var cache = CacheService.getScriptCache();
  var n = cache.get(key + '_n');
  if (!n) return null;
  var keys = [];
  for (var i = 0; i < +n; i++) keys.push(key + '_' + i);
  var got = cache.getAll(keys);
  var out = '';
  for (var k = 0; k < keys.length; k++) {
    if (got[keys[k]] == null) return null;
    out += got[keys[k]];
  }
  return out;
}

// Đọc lại Sheet và ghi cache mới NGAY (khách không phải chờ đọc Sheet)
function rebuildCache() {
  var json = JSON.stringify(readSheet_());
  putCache_(CACHE_KEY, json);
}

// Được gọi tự động khi bạn sửa Sheet (cần chạy setupTriggers một lần)
function onSheetEdit() {
  rebuildCache();
}

// ---- CHẠY HÀM NÀY MỘT LẦN để cài trigger tự động ----
function setupTriggers() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  // Xóa trigger cũ của script này để không bị trùng
  ScriptApp.getProjectTriggers().forEach(function (t) {
    var fn = t.getHandlerFunction();
    if (fn === 'onSheetEdit' || fn === 'rebuildCache' || fn === 'onEditClearCache') {
      ScriptApp.deleteTrigger(t);
    }
  });

  // 1) Sửa Sheet -> làm mới cache ngay
  ScriptApp.newTrigger('onSheetEdit').forSpreadsheet(ss).onEdit().create();

  // 2) Mỗi giờ làm mới cache một lần để luôn "ấm", khách không bao giờ gặp cache trống
  ScriptApp.newTrigger('rebuildCache').timeBased().everyHours(1).create();

  rebuildCache();
}

//  THEO DÕI LƯỢT BẤM

function textOutput_(text) {
  return ContentService.createTextOutput(text).setMimeType(ContentService.MimeType.TEXT);
}

function jsonOutput_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

// Lấy (hoặc tạo) tab ghi nhật ký lượt bấm
function getClickSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(CLICK_SHEET);
  if (!sheet) {
    sheet = ss.insertSheet(CLICK_SHEET); // thêm vào cuối nên tab sản phẩm vẫn là tab đầu tiên
    sheet.appendRow(['Thoi gian', 'Loai', 'ID san pham', 'Nguon', 'Thiet bi', 'Phien']);
    // Định dạng văn bản để Google Sheet không tự đổi "2026-11-18 09:00:00" thành ngày, hay "005" thành 5
    sheet.getRange('A:F').setNumberFormat('@');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

// Nhận 1 sự kiện từ trang web: {type:'view'|'click', id:'123', src:'tiktok', dev:'ios', sid:'abc'}
function doPost(e) {
  try {
    var raw = e && e.postData && e.postData.contents ? String(e.postData.contents) : '';
    if (!raw || raw.length > MAX_BODY_LENGTH) return textOutput_('ignored');

    var data = JSON.parse(raw);
    var type = String(data.type || '');
    if (!VALID_TYPES[type]) return textOutput_('ignored');

    // Chỉ giữ ký tự an toàn, độ dài ngắn: chặn người lạ nhét dữ liệu rác/công thức vào Sheet
    var rawId = String(data.id || '');
    var rawSid = String(data.sid || '');
    var id = rawId.replace(/[^0-9A-Za-z_\-]/g, '').slice(0, 12);
    var sid = rawSid.replace(/[^0-9A-Za-z_\-]/g, '').slice(0, 24);
    // Trang web thật chỉ gửi ký tự an toàn. Nếu có ký tự lạ hoặc quá dài thì là dữ liệu bất thường: bỏ luôn, không ghi
    if (id !== rawId || sid !== rawSid) return textOutput_('ignored');
    var src = VALID_SOURCES[data.src] ? data.src : 'other';
    var dev = VALID_DEVICES[data.dev] ? data.dev : 'other';
    if (!sid) return textOutput_('ignored');
    if (type === 'click' && !id) return textOutput_('ignored');

    // Giới hạn số sự kiện mỗi phiên
    var cache = CacheService.getScriptCache();
    var cacheKey = 'ev_' + sid;
    var count = Number(cache.get(cacheKey) || 0);
    if (count >= MAX_EVENTS_PER_SESSION) return textOutput_('limit');
    cache.put(cacheKey, String(count + 1), 21600);

    var now = Utilities.formatDate(new Date(), TIMEZONE, 'yyyy-MM-dd HH:mm:ss');
    getClickSheet_().appendRow([now, type, id, src, dev, sid]);
    return textOutput_('ok');
  } catch (err) {
    return textOutput_('error');
  }
}

// Trang kiểm tra trên máy gọi hàm này. Chỉ trả dữ liệu khi đúng khóa bí mật.
function statsResponse_(e) {
  var expected = PropertiesService.getScriptProperties().getProperty('STATS_KEY');
  var given = String((e.parameter && e.parameter.key) || '');
  if (!expected || given !== expected) {
    return jsonOutput_({ ok: false, error: 'Sai khoa bi mat hoac chua chay setupClickTracking().' });
  }

  var days = Math.max(1, Math.min(400, Number(e.parameter.days) || 120));
  var cutoff = Utilities.formatDate(new Date(Date.now() - days * 86400000), TIMEZONE, 'yyyy-MM-dd');

  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CLICK_SHEET);
  var perDay = {};

  if (sheet && sheet.getLastRow() >= 2) {
    var rows = sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).getValues();
    for (var i = 0; i < rows.length; i++) {
      var ts = rows[i][0];
      if (ts instanceof Date) ts = Utilities.formatDate(ts, TIMEZONE, 'yyyy-MM-dd HH:mm:ss');
      ts = String(ts);
      var date = ts.slice(0, 10);
      if (date < cutoff) continue;

      var type = String(rows[i][1]);
      var id = String(rows[i][2]);
      var src = String(rows[i][3]) || 'other';
      var sid = String(rows[i][5]);

      var day = perDay[date];
      if (!day) {
        day = perDay[date] = { date: date, views: 0, clicks: 0, sessions: {}, byProduct: {}, bySource: {} };
      }
      var s = day.bySource[src] || (day.bySource[src] = { views: 0, clicks: 0 });

      if (type === 'view') {
        day.views++;
        s.views++;
        if (sid) day.sessions[sid] = 1;
      } else if (type === 'click') {
        day.clicks++;
        s.clicks++;
        if (id) day.byProduct[id] = (day.byProduct[id] || 0) + 1;
      }
    }
  }

  var out = Object.keys(perDay).sort().map(function (date) {
    var d = perDay[date];
    return {
      date: d.date,
      views: d.views,
      clicks: d.clicks,
      sessions: Object.keys(d.sessions).length,
      byProduct: d.byProduct,
      bySource: d.bySource
    };
  });

  return jsonOutput_({ ok: true, generatedAt: Utilities.formatDate(new Date(), TIMEZONE, 'yyyy-MM-dd HH:mm:ss'), days: out });
}

// Xóa nhật ký quá cũ (nhật ký được ghi theo thứ tự thời gian nên các dòng cũ nằm ở trên cùng)
function cleanClickLog() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CLICK_SHEET);
  if (!sheet || sheet.getLastRow() < 2) return;

  var cutoff = Utilities.formatDate(new Date(Date.now() - CLICK_KEEP_DAYS * 86400000), TIMEZONE, 'yyyy-MM-dd');
  var times = sheet.getRange(2, 1, sheet.getLastRow() - 1, 1).getValues();
  var oldRows = 0;
  while (oldRows < times.length) {
    var t = times[oldRows][0];
    if (t instanceof Date) t = Utilities.formatDate(t, TIMEZONE, 'yyyy-MM-dd');
    if (String(t).slice(0, 10) >= cutoff) break;
    oldRows++;
  }
  if (oldRows > 0) sheet.deleteRows(2, oldRows);
}

// ---- CHẠY HÀM NÀY MỘT LẦN: tạo tab, tạo khóa bí mật, cài dọn dẹp hằng ngày ----
function setupClickTracking() {
  getClickSheet_();

  var props = PropertiesService.getScriptProperties();
  var key = props.getProperty('STATS_KEY');
  if (!key) {
    key = Utilities.getUuid().replace(/-/g, '') + Utilities.getUuid().replace(/-/g, '').slice(0, 8);
    props.setProperty('STATS_KEY', key);
  }

  // Cài trigger dọn dẹp mỗi ngày lúc 3 giờ sáng (không tạo trùng)
  var hasCleaner = ScriptApp.getProjectTriggers().some(function (t) {
    return t.getHandlerFunction() === 'cleanClickLog';
  });
  if (!hasCleaner) {
    ScriptApp.newTrigger('cleanClickLog').timeBased().everyDays(1).atHour(3).create();
  }

  console.log('Da xong. KHOA BI MAT cua ban (dan vao file config.json cua trang kiem tra): ' + key);
  return key;
}
